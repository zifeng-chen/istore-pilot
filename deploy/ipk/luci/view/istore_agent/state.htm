<% local fs = require "nixio.fs" local sys = require "luci.sys" local uci = require "luci.model.uci".cursor() local port = uci:get("istore-agent", "main", "port") or "8443" local bind = uci:get("istore-agent", "main", "bind") or "192.168.100.1" local enabled = uci:get_bool("istore-agent", "main", "enabled") -- 进程怎么找：不能再用 `pgrep -f 'istore-agent/run.py'`。 -- 这个模式会匹配到**执行 pgrep 的那个 shell**（它的命令行里就有这串字）， -- pid 回绕之后 shell 的 pid 往往比 Agent 还小，`head -1` 就挑到它了 —— -- 那个进程刚起来 0 秒，算出来的时长是负数，页面于是永远显示「—」。 -- 改成直接读 /proc//cmdline，要求「argv[1] 是 python、argv[2] 正好是 -- 那个 run.py 路径」，精确匹配，躲开自己人。 local function agent_pid() local found local ok, iter = pcall(fs.dir, "/proc") if not ok or not iter then return nil end for name in iter do local pid = tonumber(name:match("^(%d+)$")) if pid then local raw = fs.readfile("/proc/" .. pid .. " '") == 0) local version = "—" local vf = io.open("/usr/lib/istore-agent/VERSION", "r") if vf then version = vf:read("*l") or "—" vf:close() end -- ---------------------------------------------------------------- 服务运行时长 -- 原来这里用 `sys.exec("pgrep ...")` 拿到 PID 再拼文件路径， -- 但 luci.sys.exec **不会去掉结尾换行**，于是路径变成 "/proc/1234\n/stat"， -- io.open / cut 全部失败，页面上的运行时长永远显示「—」。 -- 现在直接按字节解析 /proc//stat，并且所有外部命令的输出都先裁掉空白。 local function trim(s) -- 括号很关键：string.gsub 返回两个值（结果 + 替换次数）， -- 直接 return 会被上层当成两个参数，tonumber(x, count) 就报 "base out of range" return (tostring(s or ""):gsub("^%s+", ""):gsub("%s+$", "")) end -- starttime 的单位是 USER_HZ。能问就问 getconf，但这台固件**根本没装 getconf**， -- 只能按常见值挨个验算：算出来的时长必须落在 0..系统运行时长 之间，不对就换下一个 local function clk_tck() local v = tonumber(trim(sys.exec("getconf CLK_TCK 2>/dev/null"))) if v and v > 0 then return v end return nil end local function proc_age(pid) local fh = io.open("/proc/" .. pid .. "/stat", "r") if not fh then return nil end local line = fh:read("*l") or "" fh:close() -- comm 字段带括号且可能含空格，必须从**最后一个**右括号往右切 local rest = line:match("%)%s*(.*)$") if not rest then return nil end local fields = {} for word in rest:gmatch("%S+") do fields[#fields + 1] = word end -- rest 是从第 3 个字段（state）开始的，所以第 22 个字段 starttime → fields[20] local ticks = tonumber(fields[20]) if not ticks then return nil end local boot = sys.uptime() if not boot or boot <= 0 then return nil end local hz = clk_tck() local candidates = hz and { hz } or { 100, 250, 1000 } for _, h in ipairs(candidates) do local up = math.floor(boot - ticks / h) if up > 0 and up <= boot then return up end end return nil end -- 带上秒：这一格是要「一秒一跳」的，只到分钟看着像定住了 local function fmt_span(sec) if not sec then return nil end local d = math.floor(sec / 86400) local h = math.floor((sec % 86400) / 3600) local mi = math.floor((sec % 3600) / 60) local ss = sec % 60 local hms = string.format("%02d:%02d:%02d", h, mi, ss) if d > 0 then return string.format("%d 天 %s", d, hms) end return hms end local uptime, uptime_sec = "—", nil if agent then uptime_sec = proc_age(agent) uptime = fmt_span(uptime_sec) or "—" end -- 秒数单独算成一个不含引号的字符串再输出。 -- **注释里绝对不能出现模板定界符**（小于号 + 百分号/百分号 + 大于号）： -- 解析器不认注释，会在那里就把代码块截断，后面的 HTML 全被当成 Lua 吞掉， -- 页面表现是整块内容凭空消失，而且不报错 —— 排查了很久。 local uptime_attr = uptime_sec and tostring(uptime_sec) or "" -- ---------------------------------------------------------------- 初始密码 local initial_pw = nil local ipf = io.open("/etc/istore-agent/INITIAL_PASSWORD", "r") if ipf then initial_pw = trim(ipf:read("*l") or "") ipf:close() if initial_pw == "" then initial_pw = nil end end -- ---------------------------------------------------------------- 可下载的安装包 local apk_dir = "/www/istore-agent" local apks = {} if fs.access(apk_dir) then local ok, iter = pcall(fs.dir, apk_dir) if ok then for name in iter do name = name:gsub("^%./", "") if name:match("%.apk$") then local st = fs.stat(apk_dir ..
