"/stat", "r") if not fh then return nil end local line = fh:read("*l") or "" fh:close() -- comm 字段带括号且可能含空格，必须从**最后一个**右括号往右切 local rest = line:match("%)%s*(.*)$") if not rest then return nil end local fields = {} for word in rest:gmatch("%S+") do fields[#fields + 1] = word end -- rest 是从第 3 个字段（state）开始的，所以第 22 个字段 starttime → fields[20] local ticks = tonumber(fields[20]) if not ticks then return nil end local up = math.floor(sys.uptime() - ticks / clk_tck()) if up <= 0 or up > 315360000 then return nil end return up end local function fmt_span(sec) if not sec then return nil end local d = math.floor(sec / 86400) local h = math.floor((sec % 86400) / 3600) local mi = math.floor((sec % 3600) / 60) if d > 0 then return string.format("%d 天 %d 小时", d, h) end if h > 0 then return string.format("%d 小时 %d 分", h, mi) end if mi > 0 then return string.format("%d 分钟", mi) end return string.format("%d 秒", sec % 60) end local uptime = "—" if running then local pid = trim(sys.exec("pgrep -f 'istore-agent/run.py' | head -1")) if pid ~= "" then uptime = fmt_span(proc_age(pid)) or "—" end end -- ---------------------------------------------------------------- 初始密码 local initial_pw = nil local ipf = io.open("/etc/istore-agent/INITIAL_PASSWORD", "r") if ipf then initial_pw = trim(ipf:read("*l") or "") ipf:close() if initial_pw == "" then initial_pw = nil end end -- ---------------------------------------------------------------- 可下载的安装包 local apk_dir = "/www/istore-agent" local apks = {} if fs.access(apk_dir) then local ok, iter = pcall(fs.dir, apk_dir) if ok then for name in iter do name = name:gsub("^%./", "") if name:match("%.apk$") then local st = fs.stat(apk_dir .. "/" .. name) apks[#apks + 1] = { name = name, size = st and st.size or 0, mtime = st and os.date("%Y-%m-%d %H:%M", st.mtime) or "", } end end end end table.sort(apks, function(a, b) return a.mtime > b.mtime end) %>
